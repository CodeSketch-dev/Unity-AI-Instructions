using UnityEngine;

namespace Game
{
    public interface IDigTerrain
    {
        // Carve a sphere at position; direction is the dig direction (used for brush orientation / fx).
        void Dig(Vector3 position, Vector3 direction, float radius);

        // Terrain layer index under a point (for sfx/texture choice); -1 when unknown.
        int GroundLayerAt(Vector3 position);
    }

    public static class DigMath
    {
        public const float BaseDigDuration = 3f, BaseThrowDuration = 2.5f, ExplosionRadius = 3f;

        public static float DigSize(float digSize, float mapDigSizeMultiplier) => digSize * mapDigSizeMultiplier;
        public static float DigDuration(float digSpeedMultiplier, float mapCooldownMultiplier) => BaseDigDuration / Mathf.Max(0.01f, digSpeedMultiplier) * mapCooldownMultiplier;
        public static float ThrowDuration(float throwSpeedMultiplier, float mapCooldownMultiplier) => BaseThrowDuration / Mathf.Max(0.01f, throwSpeedMultiplier) * mapCooldownMultiplier;
        public static float BlastRadius(float powerMultiplier, float mapDigSizeMultiplier) => ExplosionRadius * powerMultiplier * mapDigSizeMultiplier;
        public static double Reward(double perDig, double petCoin, double toolCoin, double mapCoin) => perDig * (petCoin + 1.0) * (toolCoin + 1.0) * mapCoin;
    }

    // Skeleton of the plugin wrapper; fill the body with the installed Digger calls. Keep edits async and capped per second.
    public sealed class DiggerTerrainAdapter : MonoBehaviour, IDigTerrain
    {
        [SerializeField] int _maxEditsPerSecond = 8;
        float _windowStart; int _editsInWindow;

        public void Dig(Vector3 position, Vector3 direction, float radius)
        {
            if (Time.unscaledTime - _windowStart > 1f) { _windowStart = Time.unscaledTime; _editsInWindow = 0; }
            if (++_editsInWindow > _maxEditsPerSecond) return;          // cap: terrain edits are expensive on low-end devices
            // DiggerMasterRuntime.ModifyAsyncBuffured(position, BrushType.Sphere, ActionType.Dig, textureIndex, opacity, radius, ...);
        }

        public int GroundLayerAt(Vector3 position)
        {
            // SphereCast radius 0.3 downward, map the hit to a terrain layer index.
            return -1;
        }
    }
}
