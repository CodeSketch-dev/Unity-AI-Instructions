using System.Collections.Generic;
using System.Diagnostics;
using System.IO;
using System.Text;
using UnityEditor;
using UnityEngine;
using UnityEngine.UI;

namespace Game
{
    // Automatic UI audit for the WHOLE game (skill ui-ink-audit). In Play Mode it watches what the player can see: the open popups (children of "PopupRoot"),
    // or, with no popup, the set of active buttons (HUD, shops, minigame screens, scene changes). Whenever that set changes and settles it dumps the screen,
    // captures a screenshot and runs ink_audit.py; every violation is a Console warning "[InkAudit] FAIL ..." (once per screen state) and is appended to
    // Temp/InkAudit/report.txt. Any manual or scripted Play test therefore surfaces misaligned icon+text, titles off their colour band and overlaps
    // without anybody asking. Needs `python` with Pillow on PATH; otherwise it logs once and stays quiet.
    [InitializeOnLoad]
    public static class Qa_InkWatcher
    {
        const int SETTLE_FRAMES = 25, SHOT_WAIT_FRAMES = 3;
        const double MIN_SECONDS_BETWEEN = 3.0;
        static string _signature = "", _pending = "";
        static int _stableFrames, _shotCountdown = -1;
        static double _lastAudit;
        static bool _pythonWarned;
        static string _dir, _png, _json, _label;
        static readonly HashSet<string> s_reported = new HashSet<string>();

        static Qa_InkWatcher()
        {
            EditorApplication.update += Tick;
            EditorApplication.playModeStateChanged += s => { _signature = _pending = ""; _stableFrames = 0; _shotCountdown = -1; s_reported.Clear(); };
        }

        // popup mode: names of open popups; otherwise: sorted names of visible buttons (so a scene change or a new HUD element re-triggers the audit).
        static string Signature(out string label)
        {
            var popups = new StringBuilder();
            foreach (Transform t in Object.FindObjectsByType<Transform>(FindObjectsInactive.Exclude, FindObjectsSortMode.None))
                if (t.name == "PopupRoot") for (int i = 0; i < t.childCount; i++) if (t.GetChild(i).gameObject.activeInHierarchy) popups.Append(t.GetChild(i).name).Append('|');
            if (popups.Length > 0) { label = popups.ToString(); return "P:" + label; }
            var names = new List<string>();
            foreach (Selectable s in Object.FindObjectsByType<Selectable>(FindObjectsInactive.Exclude, FindObjectsSortMode.None)) if (s.isActiveAndEnabled) names.Add(s.name);
            names.Sort();
            label = "HUD:" + UnityEngine.SceneManagement.SceneManager.GetActiveScene().name;
            return label + ":" + string.Join(",", names);
        }

        static void Tick()
        {
            if (!EditorApplication.isPlaying || EditorApplication.isPaused) return;
            if (_shotCountdown >= 0)
            {
                if (--_shotCountdown == 0) { _shotCountdown = -1; Audit(); }
                return;
            }
            string label;
            string sig = Signature(out label);
            if (sig != _pending) { _pending = sig; _stableFrames = 0; return; }
            if (++_stableFrames < SETTLE_FRAMES || sig == _signature) return;
            if (EditorApplication.timeSinceStartup - _lastAudit < MIN_SECONDS_BETWEEN) return;
            _signature = sig; _label = label; _lastAudit = EditorApplication.timeSinceStartup;
            _dir = Path.GetFullPath(Path.Combine(Application.dataPath, "..", "Temp", "InkAudit"));
            Directory.CreateDirectory(_dir);
            _png = Path.Combine(_dir, "auto.png"); _json = Path.Combine(_dir, "auto.json");
            if (File.Exists(_png)) File.Delete(_png);
            Qa_InkDump.Run(_json);
            ScreenCapture.CaptureScreenshot(_png);
            _shotCountdown = SHOT_WAIT_FRAMES;
        }

        static void Audit()
        {
            string script = Path.GetFullPath(Path.Combine(Application.dataPath, "..", ".claude", "skills", "ui-ink-audit", "ink_audit.py"));
            if (!File.Exists(script) || !File.Exists(_png)) return;
            try
            {
                var psi = new ProcessStartInfo("python", "-I \"" + script + "\" \"" + _png + "\" \"" + _json + "\" --fixes \"" + Path.Combine(_dir, "fixes.json") + "\"")
                { UseShellExecute = false, RedirectStandardOutput = true, RedirectStandardError = true, CreateNoWindow = true };
                using (Process p = Process.Start(psi))
                {
                    string output = p.StandardOutput.ReadToEnd();
                    if (!p.WaitForExit(20000)) { p.Kill(); return; }
                    int fails = 0;
                    var report = new StringBuilder();
                    foreach (string raw in output.Split('\n'))
                    {
                        string line = raw.Trim();
                        if (!line.StartsWith("FAIL")) continue;
                        fails++; report.AppendLine(line);
                        if (s_reported.Add(line)) UnityEngine.Debug.LogWarning("[InkAudit] " + _label + " " + line);   // each distinct violation logged once per session
                    }
                    UnityEngine.Debug.Log("[InkAudit] " + _label + " -> " + (fails == 0 ? "0 fail" : fails + " FAIL"));
                    File.AppendAllText(Path.Combine(_dir, "report.txt"), System.DateTime.Now.ToString("HH:mm:ss") + " " + _label + " " + fails + " fail\n" + report);
                }
            }
            catch (System.Exception e)
            {
                if (_pythonWarned) return;
                _pythonWarned = true;
                UnityEngine.Debug.LogWarning("[InkAudit] python/Pillow not available, auto UI audit disabled: " + e.Message);
            }
        }
    }
}
