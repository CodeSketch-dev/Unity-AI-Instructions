// Serialized-reference example for an authored uGUI prefab. No runtime hierarchy construction.
// Adapt its base/lifetime to the installed framework. Compile and run the skill's Play matrix after adoption.
// FocusDimBox/Focus and FocusHandBox are direct children of a full-canvas Root.
using UnityEngine;
using UnityEngine.UI;

public sealed class QuestFocusPointer : MonoBehaviour
{
    [SerializeField] RectTransform _root, _dimBox, _dimFocus, _handBox, _hand;
    [SerializeField] Canvas _overlayCanvas;
    [SerializeField] Image[] _dimPanels;
    [SerializeField] Graphic[] _decorations; // Exclude an interactive notice's owning Image.
    [SerializeField] Button _noticeButton; // Optional real notice Button, with project press/sfx components.
    [SerializeField] float _offsetValue = 1000f, _duration = .35f, _dimAlpha = .9607843f;
    [SerializeField] Vector2 _handAnchor = new Vector2(.8f, .3f);
    [SerializeField] Vector2 _handPivot = new Vector2(.1f, .78f);
    [SerializeField] Vector2 _handSize = new Vector2(134.35f, 127f);
    [SerializeField] float _handScale = 1.1f, _handHalfPeriod = .5f;

    readonly Vector3[] _corners = new Vector3[4];
    RectTransform _target;
    Canvas _targetCanvas;
    Button _targetAction, _boundAction;
    float _appear;
    bool _showing;

    void Awake()
    {
        foreach (Graphic graphic in _decorations) if (graphic != null) graphic.raycastTarget = false;
        foreach (Image panel in _dimPanels) if (panel != null) panel.raycastTarget = false;
        _hand.anchorMin = _hand.anchorMax = _handAnchor;
        _hand.pivot = _handPivot;
        _hand.sizeDelta = _handSize;
        _hand.anchoredPosition = Vector2.zero;
        _hand.localRotation = Quaternion.identity;
        _dimBox.SetAsFirstSibling();
        _handBox.SetAsLastSibling();
    }

    // Call on target/step change, not each frame. A custom control completes via its own accepted-action event.
    public void Show(RectTransform target, Button action = null)
    {
        if (_showing && target == _target && isActiveAndEnabled) return;
        UnbindAction();
        _target = target;
        _targetAction = action != null ? action : target != null ? target.GetComponent<Button>() : null;
        _targetCanvas = target != null ? target.GetComponentInParent<Canvas>() : null;
        _appear = 0f;
        _showing = target != null;
        gameObject.SetActive(_showing);
        if (isActiveAndEnabled) BindAction();
    }

    public void Hide()
    {
        _showing = false;
        UnbindAction();
        _target = null;
        _targetAction = null;
        _targetCanvas = null;
        gameObject.SetActive(false);
    }

    // Only the notice Button calls this. The target itself receives its original EventSystem gesture.
    void OnNoticeClick()
    {
        if (_targetAction == null || !_targetAction.IsActive() || !_targetAction.IsInteractable()) return;
        _targetAction.onClick.Invoke();
    }

    // A custom hold/drag owner can wire its real accepted-action event here without inventing a click.
    public void CompleteAcceptedAction() => Hide();

    void BindAction()
    {
        if (_targetAction == null || _boundAction == _targetAction) return;
        _boundAction = _targetAction;
        _boundAction.onClick.AddListener(CompleteAcceptedAction);
    }

    void UnbindAction()
    {
        if (_boundAction != null) _boundAction.onClick.RemoveListener(CompleteAcceptedAction);
        _boundAction = null;
    }

    void OnEnable()
    {
        if (_showing) BindAction();
        if (_noticeButton != null) _noticeButton.onClick.AddListener(OnNoticeClick);
    }

    void OnDisable()
    {
        UnbindAction();
        if (_noticeButton != null) _noticeButton.onClick.RemoveListener(OnNoticeClick);
    }

    void OnDestroy() => UnbindAction();

    static Camera CanvasCamera(Canvas canvas)
    {
        if (canvas == null) return null;
        Canvas root = canvas.rootCanvas;
        return root.renderMode == RenderMode.ScreenSpaceOverlay ? null : root.worldCamera;
    }

    void LateUpdate()
    {
        if (!_showing || _target == null || !_target.gameObject.activeInHierarchy) { Hide(); return; }
        Camera sourceCamera = CanvasCamera(_targetCanvas);
        Camera overlayCamera = CanvasCamera(_overlayCanvas);
        _target.GetWorldCorners(_corners);
        Vector2 min = new Vector2(float.PositiveInfinity, float.PositiveInfinity);
        Vector2 max = new Vector2(float.NegativeInfinity, float.NegativeInfinity);
        for (int i = 0; i < _corners.Length; i++)
        {
            Vector2 screen = RectTransformUtility.WorldToScreenPoint(sourceCamera, _corners[i]);
            if (!RectTransformUtility.ScreenPointToLocalPointInRectangle(_root, screen, overlayCamera, out Vector2 local)) { Hide(); return; }
            min = Vector2.Min(min, local);
            max = Vector2.Max(max, local);
        }
        Vector2 center = (min + max) * .5f;
        Vector2 size = max - min;
        _dimBox.anchoredPosition = _handBox.anchoredPosition = center;
        _dimBox.sizeDelta = _handBox.sizeDelta = size;
        _appear = Mathf.Min(1f, _appear + Time.unscaledDeltaTime / Mathf.Max(.01f, _duration));
        float eased = Mathf.Sin(_appear * Mathf.PI * .5f);
        float offset = _offsetValue * (1f - eased);
        _dimFocus.offsetMin = Vector2.one * -offset;
        _dimFocus.offsetMax = Vector2.one * offset;
        foreach (Image panel in _dimPanels)
        {
            if (panel == null) continue;
            Color color = panel.color;
            color.a = _dimAlpha * eased;
            panel.color = color;
        }
        float phase = Mathf.PingPong(Time.unscaledTime / Mathf.Max(.01f, _handHalfPeriod), 1f);
        _hand.localScale = Vector3.one * Mathf.Lerp(1f, _handScale, Mathf.SmoothStep(0f, 1f, phase));
        if (_noticeButton != null)
            _noticeButton.interactable = _targetAction != null && _targetAction.IsActive() && _targetAction.IsInteractable();
    }
}
