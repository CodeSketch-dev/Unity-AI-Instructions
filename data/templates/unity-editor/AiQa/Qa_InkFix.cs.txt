using System.Collections.Generic;
using System.IO;
using TMPro;
using UnityEditor;
using UnityEngine;
using UnityEngine.UI;

namespace AiQa
{
    // Applies the corrections ink_audit.py wrote to Temp/InkAudit/fixes.json (only items that FAILED; aligned UI is never touched) to the SOURCE prefabs.
    // Edit Mode only. Button with a HorizontalLayoutGroup: padding right/bottom grow by 2*d/scale (the group centres, so the pair moves by d). No group:
    // the icon/label children move by -d. Title: the label moves by -dy. Run again in Play afterwards: the watcher must report 0 fail.
    public static class Qa_InkFix
    {
        [System.Serializable] sealed class Fix { public string kind, prefab, rel; public float dx, dy, scale; }
        [System.Serializable] sealed class FixList { public List<Fix> fixes = new List<Fix>(); }

        [MenuItem("Tools/Qa/Apply Ink Fixes")]
        public static void RunMenu() => Debug.Log(Run());

        public static string Run(bool allowInPlay = false)
        {
            if (EditorApplication.isPlaying && !allowInPlay) return "[InkFix] stop Play Mode first";
            string file = Path.GetFullPath(Path.Combine(Application.dataPath, "..", "Temp", "InkAudit", "fixes.json"));
            if (!File.Exists(file)) return "[InkFix] no fixes.json";
            var list = JsonUtility.FromJson<FixList>(File.ReadAllText(file));
            var byPrefab = new Dictionary<string, List<Fix>>();
            int skipped = 0;
            // Safety: a given target is auto-corrected at most twice; if it still fails after that it needs a human look (logged, never loops).
            string attemptsFile = Path.GetFullPath(Path.Combine(Application.dataPath, "..", "Library", "InkFixAttempts.txt"));
            var attempts = new Dictionary<string, int>();
            if (File.Exists(attemptsFile))
                foreach (string line in File.ReadAllLines(attemptsFile)) { int i = line.LastIndexOf('='); if (i > 0 && int.TryParse(line.Substring(i + 1), out int n)) attempts[line.Substring(0, i)] = n; }
            foreach (Fix f in list.fixes)
            {
                string key = f.prefab + "|" + f.rel + "|" + f.kind;
                attempts.TryGetValue(key, out int done);
                if (done >= 2) { Debug.LogWarning("[InkFix] gave up after 2 automatic attempts, needs a manual look: " + key); skipped++; continue; }
                attempts[key] = done + 1;
                if (string.IsNullOrEmpty(f.prefab)) { skipped++; continue; }
                if (!byPrefab.TryGetValue(f.prefab, out var l)) byPrefab[f.prefab] = l = new List<Fix>();
                l.Add(f);
            }
            int applied = 0;
            foreach (var kv in byPrefab)
            {
                GameObject root = PrefabUtility.LoadPrefabContents(kv.Key);
                try
                {
                    foreach (Fix f in kv.Value)
                    {
                        Transform t = string.IsNullOrEmpty(f.rel) ? root.transform : root.transform.Find(f.rel);
                        if (t == null) { skipped++; continue; }
                        float sc = f.scale > 0.01f ? f.scale : 1f;
                        if (f.kind == "title") Shift(t, 0f, -f.dy / sc);
                        else ApplyButton(t, f.dx / sc, f.dy / sc);
                        applied++;
                    }
                    PrefabUtility.SaveAsPrefabAsset(root, kv.Key);
                }
                finally { PrefabUtility.UnloadPrefabContents(root); }
            }
            var lines = new List<string>(); foreach (var kv in attempts) lines.Add(kv.Key + "=" + kv.Value);
            File.WriteAllLines(attemptsFile, lines);
            File.Delete(file);
            return "[InkFix] applied " + applied + " fix(es) to " + byPrefab.Count + " prefab(s), skipped " + skipped + " (runtime-built or missing)";
        }

        static void Shift(Transform t, float dx, float dyUp) { var rt = (RectTransform)t; rt.anchoredPosition += new Vector2(dx, dyUp); }

        const float MAX_STEP = 8f, MAX_PADDING = 16f;   // a correction larger than this is a measurement problem, not a layout problem: leave it for a human

        static void ApplyButton(Transform button, float dx, float dyDown)
        {
            if (Mathf.Abs(dx) > 20f || Mathf.Abs(dyDown) > 20f) { Debug.LogWarning("[InkFix] implausible correction (" + dx.ToString("F0") + "," + dyDown.ToString("F0") + ") on " + button.name + ", needs a manual look"); return; }
            // ink lower than the face centre by dyDown (units) -> raise the pair; ink right by dx -> move it left
            var hlg = button.GetComponent<HorizontalLayoutGroup>() ?? button.GetComponentInChildren<HorizontalLayoutGroup>(true);
            if (hlg != null)
            {
                RectOffset p = hlg.padding;
                int sx = Mathf.RoundToInt(Mathf.Clamp(2f * dx, -MAX_STEP, MAX_STEP)), sy = Mathf.RoundToInt(Mathf.Clamp(2f * dyDown, -MAX_STEP, MAX_STEP));
                if (sy > 0) p.bottom = Mathf.Min(Mathf.RoundToInt(MAX_PADDING), p.bottom + sy); else p.top = Mathf.Min(Mathf.RoundToInt(MAX_PADDING), p.top - sy);
                if (sx > 0) p.right = Mathf.Min(Mathf.RoundToInt(MAX_PADDING), p.right + sx); else p.left = Mathf.Min(Mathf.RoundToInt(MAX_PADDING), p.left - sx);
                hlg.padding = p;
                return;
            }
            foreach (Transform c in button)
                if (c.GetComponent<TMP_Text>() != null || (c.GetComponent<Image>() != null && c.name != "Background" && c.name != "Frame")) Shift(c, -dx, dyDown);
        }
    }
}
